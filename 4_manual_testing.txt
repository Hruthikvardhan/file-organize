"""Creates ~/organizer_test with sample files for manual testing."""
import os
import time
from pathlib import Path

root = Path.home() / "organizer_test"
root.mkdir(exist_ok=True)


def make(name, size=10, days_old=0, content=None):
    """Create a file; optionally backdate its modified time."""
    path = root / name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(content if content is not None else os.urandom(size))
    if days_old:
        stamp = time.time() - days_old * 86400
        os.utime(path, (stamp, stamp))


# One or more files for every default category
for name in ["photo1.jpg", "photo2.PNG", "logo.svg", "clip.mp4", "song.mp3",
             "report.pdf", "notes.txt", "data.csv", "script.py", "page.html",
             "backup.zip", "setup.exe", "mystery.xyz", "README"]:
    make(name)

# Three identical files with different modified dates (duplicate finder)
same = b"identical content for duplicate test"
make("dup_a.txt", content=same, days_old=5)   # oldest
make("dup_b.txt", content=same, days_old=2)
make("dup_c.txt", content=same)               # newest

make("old_photo.jpg", days_old=400)           # date mode: older year
make("medium_video.mkv", size=2 * 1024 * 1024)  # size mode: Medium (2 MB)
make("empty_file.txt", size=0)                # empty files are never duplicates
make("empty_file2.txt", size=0)

# Files that must be SKIPPED
make(".hidden_file", size=5)                  # hidden
make("download.crdownload")                   # unfinished download
make("node_modules/lib.js")                   # excluded folder

# A nested file (only moved with --recursive)
make("subfolder/nested.docx")

print("Created test folder:", root)
